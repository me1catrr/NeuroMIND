% Figura de UNA columna
% Uso: \figcol[etiqueta]{ruta/figura}{texto de pie}
\newcommand{\figcol}[3][]{%
  \begin{figure}[!htbp]
    \centering
    \includegraphics[width=\columnwidth]{#2}%
    \caption{#3}%
    \ifx\relax#1\relax\else\label{#1}\fi
  \end{figure}%
}

% Figura a DOBLE columna (usa figure* en modo twocolumn)
\makeatletter
\newcommand{\figwide}[3][]{%
  \if@twocolumn
    \begin{figure*}[t]
      \centering
      \includegraphics[width=\textwidth]{#2}%
      \caption{#3}%
      \ifx\relax#1\relax\else\label{#1}\fi
    \end{figure*}%
  \else
    \begin{figure}[!htbp]
      \centering
      \includegraphics[width=\textwidth]{#2}%
      \caption{#3}%
      \ifx\relax#1\relax\else\label{#1}\fi
    \end{figure}%
  \fi
}
\makeatother

% Tabla de UNA columna ajustada al ancho
% Uso: \tabcol[etiqueta]{texto de pie}{<tabular>}
\newcommand{\tabcol}[3][]{%
  \begin{table}[!htbp]
    \centering
    \caption{#2}%
    \ifx\relax#1\relax\else\label{#1}\fi
    \begin{adjustbox}{max width=\columnwidth}
      #3%
    \end{adjustbox}
  \end{table}%
}

% Tabla a DOBLE columna (usa table* en modo twocolumn)
\makeatletter
\newcommand{\tabwide}[3][]{%
  \if@twocolumn
    \begin{table*}[t]
      \centering
      \caption{#2}%
      \ifx\relax#1\relax\else\label{#1}\fi
      #3%
    \end{table*}%
  \else
    \begin{table}[!htbp]
      \centering
      \caption{#2}%
      \ifx\relax#1\relax\else\label{#1}\fi
      \begin{adjustbox}{max width=\textwidth}
        #3%
      \end{adjustbox}
    \end{table}%
  \fi
}
\makeatother
